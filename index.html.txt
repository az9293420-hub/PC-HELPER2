// ==================== КОНФИГУРАТОР ====================
function fillSelect(id, items, keepValue) {
  const el = document.getElementById(id);
  const prev = keepValue ? el.value : null;
  el.innerHTML = items.map(x => `<option value="${x[0]}">${x[1]} — ${fmt(x[2])}</option>`).join('');
  if (prev && items.some(x => x[0] === prev)) el.value = prev;
}

function syncSocket(sid) {
  fillSelect('cpu', CPUs[sid] || []);
  fillSelect('mb', MB[sid] || []);
  fillSelect('ram', RAM[sid] || []);
  render();
}

function render() {
  const sid = document.getElementById('socket').value;
  const cpu = find(CPUs[sid]||[], document.getElementById('cpu').value);
  const gpu = find(GPUs, document.getElementById('gpu').value);
  const mb  = find(MB[sid]||[], document.getElementById('mb').value);
  const ram = find(RAM[sid]||[], document.getElementById('ram').value);
  const ssd = find(SSD, document.getElementById('ssd').value);
  const psu = find(PSU, document.getElementById('psu').value);
  const cse = find(CASES, document.getElementById('case').value);
  if (!cpu || !gpu || !mb || !ram || !ssd || !psu || !cse) return;

  let total = cpu[2] + gpu[2] + mb[2] + ram[2] + ssd[2] + psu[2] + cse[2];

  let rows = `
    <ul class="comp">
      <li><span>${cpu[1]}</span><strong>${fmt(cpu[2])}</strong></li>
      <li><span>${gpu[1]}</span><strong>${gpu[2]?fmt(gpu[2]):'—'}</strong></li>
      <li><span>${mb[1]}</span><strong>${fmt(mb[2])}</strong></li>
      <li><span>${ram[1]}</span><strong>${fmt(ram[2])}</strong></li>
      <li><span>${ssd[1]}</span><strong>${fmt(ssd[2])}</strong></li>
      <li><span>${psu[1]}</span><strong>${fmt(psu[2])}</strong></li>
      <li><span>${cse[1]}</span><strong>${fmt(cse[2])}</strong></li>
    </ul>`;

  if (gpu[0] !== 'none') rows += `<p>Ожидаемый ФПС (1080p / 1440p): <b>${gpu[4]}</b></p>`;

  const q = name => encodeURIComponent(name);
  rows += `<p>
    <a class="btn small" href="https://market.yandex.ru/search?text=${q(cpu[1])}" target="_blank">Я.Маркет</a>
    <a class="btn small" href="https://www.dns-shop.ru/search/?q=${q(cpu[1])}" target="_blank">DNS</a>
    <a class="btn small" href="https://www.ozon.ru/search/?text=${q(cpu[1])}" target="_blank">Ozon</a>
  </p>`;

  document.getElementById('configBox').innerHTML = rows;
  document.getElementById('total').innerHTML = `💰 Общая цена сборки: <b>${fmt(total)}</b>`;

  let warn = '';

}

  let ssd = find(SSD,'ssd_1t');
  const psus = PSU.filter(p => p[3] >= gpu[3] + 150);
  if (!psus.length) return null;
  const psu = psus.reduce((a,b) => a[2] < b[2] ? a : b);
  let cse = find(CASES,'m6');
  const tot = () => gpu[2] + cpu[2] + mb[2] + ram[2] + ssd[2] + psu[2] + cse[2];
  if (tot() > budget) { ssd = find(SSD,'ssd_512'); if (tot() > budget) { ssd = find(SSD,'ssd_240'); if (tot() > budget) return null; } }
  const up = (cur, list) => { let best = cur; for (const o of list) if (o[2] > best[2] && tot() - cur[2] + o[2] <= budget) best = o; return best; };
  ssd = up(ssd, SSD); ram = up(ram, ramList); cse = up(cse, CASES); mb = up(mb, MB[sid]);
  return {sid, gpu, cpu, mb, ram, ssd, psu, case:cse, total: tot()};
}

function pickByBudget() {
  const budget = Number(document.getElementById('budgetIn').value);
  const msg = document.getElementById('budgetMsg');
  if (!budget || budget < 20000) { msg.innerHTML = '<div class="warn">Минимальная стоимость сборки — 20 000 ₽.</div>'; return; }
  msg.innerHTML = '<div class="ok">Подбираю…</div>';

  let best = null, bestKey = null;
  for (const g of GPUs) {
    if (g[0] === 'none') continue;
    for (const sid of PLAT_ORDER) {
      for (const c of CPUs[sid]) {
        if (g[2] > budget || c[2] > budget) continue;
        if (!(0.15 * g[2] <= c[2] && c[2] <= 2.5 * g[2])) continue; // баланс ЦП/видеокарта
        if (g[5] - c[3] > 6) continue;                              // процессор не старше карты на 6+ лет
        const r = buildCand(sid, g, c, budget);
        if (!r) continue;
        const key = [GPU_RANK[g[0]], PLAT_ORDER.indexOf(sid), CPU_RANK[c[0]]];
        if (!bestKey || key > bestKey) { bestKey = key; best = r; }
      }
    }
    if (best && best.total <= budget) break;
  }

  if (!best) { // офисный фолбэк на встроенной графике
    for (const sid of ['am4','l1200','am3p']) {
      const cpuId = sid === 'am4' ? 'r5600g' : sid === 'l1200' ? 'i511400' : null;
      if (!cpuId) continue;
      const c = find(CPUs[sid], cpuId);
      const r = buildCand(sid, find(GPUs,'none'), c, budget);
      if (r) { best = r; break; }
    }
  }
  if (!best) { msg.innerHTML = '<div class="warn">В этот бюджет ничего не влезает. Попробуйте сумму побольше.</div>'; return; }

  const fps = best.gpu[4] !== '-' ? ` · ФПС ≈ ${best.gpu[4]}` : '';
  msg.innerHTML = `<div class="card">Сборка: <b>${best.gpu[1]}</b> + <b>${best.cpu[1]}</b> + ${best.mb[1]} + ${best.ram[1]} + ${best.ssd[1]} + ${best.psu[1]} + ${best.case[1]}<br>
    <div class="price-big">💰 Общая цена: <b>${fmt(best.total)}</b>${fps}</div>
    <button class="btn" onclick='loadBuild(LAST_BUILD)'>Загрузить в конфигуратор</button></div>`;
  window.LAST_BUILD = best;
}

function chatAdd(text, who) {
  const log = document.getElementById('chatLog');
  log.innerHTML += `<div class="${who}">${who === 'user' ? 'Вы: ' : '🤖: '}${text}</div>`;

function sendChat() {
  const inp = document.getElementById('chatIn');
  const t = inp.value.trim();
  if (!t) return;
  chatAdd(t, 'user');
  inp.value = '';
  setTimeout(() => chatAdd(aiAnswer(t), 'bot'), 200);
}
